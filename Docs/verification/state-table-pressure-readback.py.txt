def read(p):
 fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
 try:
  st=os.fstat(fd);assert stat.S_ISREG(st.st_mode) and st.st_size<=32_000_000
  data=b''
  while True:
   part=os.read(fd,min(65536,32_000_001-len(data)))
   if not part:break
   data+=part
   assert len(data)<=32_000_000
  return data
 finally:os.close(fd)
