

extension HomeViewUITests {
  private func verifyNativeAnswerCardReading(_ scene:String,example:String,source:String?=nil,required:[[String]])throws {
    continueAfterFailure=true;executionTimeAllowance=240
    guard #available(iOS 27.0,*) else {XCTFail("Native VoiceOver service requires iOS27");return}
    let service=XCUIDevice.shared.voiceOverService,original=service.isEnabled
    let started=ProcessInfo.processInfo.systemUptime
    var events:[[String:Any]]=[],seen=Set<Int>(),restored=false
    func normalized(_ text:String)->String {
      text.lowercased().components(separatedBy:CharacterSet.alphanumerics.inverted).joined()
    }
    func record(_ event:String,_ fields:[String:Any]=[:]) {
      var row=fields;row["event"]=event;row["elapsed"]=ProcessInfo.processInfo.systemUptime-started;events.append(row)
    }
    func observe(_ event:String,_ utterance:String) {
      record(event,["utterance":utterance])
      for (index,group) in required.enumerated() where group.allSatisfy({normalized(utterance).contains(normalized($0))}) {seen.insert(index)}
    }
    defer {
      do {
        if service.isEnabled != original {if original {try service.enable()} else {try service.disable()}}
        restored=service.isEnabled==original
        record("restored-original",["enabled":service.isEnabled,"matches_original":restored]);XCTAssertTrue(restored)
      } catch {let e=error as NSError;record("restore-error",["domain":e.domain,"code":e.code]);XCTFail("Restore original owned-simulator VoiceOver state")}
      let report:[String:Any]=["scene":scene,"example":example,"source_sheet":source ?? "none",
        "required_full_literal_groups":required,"found_groups":seen.sorted(),
        "missing_groups":required.enumerated().filter{!seen.contains($0.offset)}.map{["index":$0.offset,"values":$0.element] as [String:Any]},
        "original_enabled":original,"restored":restored,"events":events,
        "scope":"Actual native reported-utterance reading, full declared short literals/field groups and local focus return. Ordinary fixture routing is not VO activation. No raw-touch/keyboard activation, acoustic/hardware or original-order acceptance. Test-only reference viewport marker retained in stream." ]
      let a=XCTAttachment(data:try! JSONSerialization.data(withJSONObject:report,options:[.sortedKeys]),uniformTypeIdentifier:"public.json")
      a.name="native-answer-card-reading-"+scene;a.lifetime = .keepAlways;add(a)
    }
    record("original-state",["enabled":original])
    // Ordinary routing deliberately occurs with the service disabled, then we
    // use only the actual native service for the declared reading traversal.
    if service.isEnabled {try service.disable()}
    XCTAssertFalse(service.isEnabled)
    launch(["--persistence-test",UUID().uuidString,"--reset-test-store"]);tap("nav.assistant")
    let draft="temperature 91",field=app.textFields["assistant.input"]
    field.tap();field.typeText(draft);selectAnswerExample(example)
    if let source {
      activatePlanControl(app.buttons["assistant.card.source."+source])
      guard app.buttons["assistant.card.source.done"].waitForExistence(timeout:5) else {XCTFail("Local source sheet did not open");return}
      XCTAssertEqual(responseElement("assistant.card.source.open").exists,source=="museum")
      shot("native-card-reading-"+scene+"-before")
    } else {
      guard app.staticTexts["assistant.card.scope"].waitForExistence(timeout:5) else {XCTFail("Declared bundled card absent before native reading");return}
      viewportShot("native-card-reading-"+scene+"-before",assistantRectangle:true)
    }
    try service.enable();record("native-enabled",["enabled":service.isEnabled])
    guard service.isEnabled else {XCTFail("Native VoiceOver not enabled");return}
    if let initial=try? service.currentSpeech().utterance {observe("initial-speech",initial)}
    let deadline=ProcessInfo.processInfo.systemUptime+90
    for _ in 0..<112 {
      if seen.count==required.count {break}
      guard ProcessInfo.processInfo.systemUptime<deadline else {break}
      observe("move-forward",try service.moveForward().utterance)
    }
    record("reading-result",["required_groups":required.count,"found_groups":seen.count])
    guard seen.count==required.count else {XCTFail("Native reading missed full declared values for \(scene)");return}
    let anchor=try service.currentSpeech().utterance
    observe("move-backward",try service.moveBackward().utterance)
    let returned=try service.moveForward().utterance;observe("move-forward-return",returned)
    XCTAssertEqual(normalized(returned),normalized(anchor),"Return to the exact same focused utterance")
    if source != nil {shot("native-card-reading-"+scene+"-focused")}
    else {viewportShot("native-card-reading-"+scene+"-focused",assistantRectangle:true)}
    // Restore an ordinary-input routing pose explicitly; none of the next
    // actions is claimed as a VoiceOver activation.
    try service.disable();XCTAssertFalse(service.isEnabled)
    if source != nil {tap("assistant.card.source.done")}
    XCTAssertEqual(field.value as? String,draft,"Reading/source display must not submit or lose the draft")
    XCTAssertTrue(app.staticTexts["assistant.card.scope"].exists)
    tap("nav.faces");tap("home.thermostat")
    XCTAssertEqual(app.staticTexts["thermostat.value"].label,"68°","Reading must not execute the unsubmitted command")
  }
  func testNativeAnswerCardKnowledgeFullReadingAndRestore()throws {

Light-sensitive iron salts produce its distinctive blue prints."],["Illustration: pale fern-like branches on a deep-blue background"],["HomeView illustration · Not a historical print"],["Source: Science Museum Group","Button"],["Sample agenda","Button"],["Recipe card","Button"]])
  }
