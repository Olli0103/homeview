import UIKit

@main final class AppDelegate: UIResponder, UIApplicationDelegate {
  func application(_ application: UIApplication, configurationForConnecting session: UISceneSession,
    options: UIScene.ConnectionOptions) -> UISceneConfiguration {
    let configuration=UISceneConfiguration(name:"Probe",sessionRole:session.role)
    configuration.delegateClass=SceneDelegate.self
    return configuration
  }
}
final class SceneDelegate: UIResponder, UIWindowSceneDelegate {
  var window: UIWindow?
  func scene(_ scene: UIScene,willConnectTo session: UISceneSession,options: UIScene.ConnectionOptions) {
    guard let scene=scene as? UIWindowScene else { return }
    let window=UIWindow(windowScene:scene)
    window.rootViewController=Control();window.makeKeyAndVisible();self.window=window
  }
}
final class Control: UIViewController, UITextFieldDelegate {
  let field=UITextField(),status=UILabel()
  var returns=0,cancels=0
  override func viewDidLoad() {
    super.viewDidLoad();view.backgroundColor = .white
    field.borderStyle = .roundedRect;field.placeholder="Synthetic control input"
    field.accessibilityIdentifier="control.input";field.returnKeyType = .send;field.delegate=self
    status.accessibilityIdentifier="control.status";status.textColor = .black;status.numberOfLines=0
    let stack=UIStackView(arrangedSubviews:[field,status]);stack.axis = .vertical;stack.spacing=24
    stack.translatesAutoresizingMaskIntoConstraints=false;view.addSubview(stack)
    NSLayoutConstraint.activate([stack.leadingAnchor.constraint(equalTo:view.safeAreaLayoutGuide.leadingAnchor,constant:24),
      stack.trailingAnchor.constraint(equalTo:view.safeAreaLayoutGuide.trailingAnchor,constant:-24),
      stack.topAnchor.constraint(equalTo:view.safeAreaLayoutGuide.topAnchor,constant:80),field.heightAnchor.constraint(equalToConstant:48)])
    refresh()
  }
  override var keyCommands: [UIKeyCommand]? {
    let escape=UIKeyCommand(input:UIKeyCommand.inputEscape,modifierFlags:[],action:#selector(cancelInput))
    escape.wantsPriorityOverSystemBehavior=true
    return [escape]
  }
  @objc func cancelInput() {cancels+=1;field.resignFirstResponder();refresh()}
  func textFieldShouldReturn(_ textField: UITextField) -> Bool {returns+=1;refresh();return false}
  func refresh() {status.text="returns=\(returns);cancels=\(cancels)"}
}
